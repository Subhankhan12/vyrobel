<?php
// READ-ONLY, NO DATABASE: from the STAGED release (stdin). Compiles every Blade view in memory and token-parses the PHP it makes, loads the new classes,
// checks the new routes and a sample of the new texts in every language, and reads the NAME of the active mailer (email must still be off).
// Writes nothing; prints counts and yes/no only — never a value, never an address.
use Illuminate\Support\Facades\Lang;
use Illuminate\Support\Facades\Route;

require getcwd().'/vendor/autoload.php';
$app = require getcwd().'/bootstrap/app.php';
$app->make(Illuminate\Contracts\Console\Kernel::class)->bootstrap();

$out = ['release' => substr(trim((string) @file_get_contents(getcwd().'/.release')), 0, 8), 'php' => PHP_VERSION];
$compiler = $app->make('blade.compiler');
$n = 0; $bad = [];
$it = new RecursiveIteratorIterator(new RecursiveDirectoryIterator(getcwd().'/resources/views', FilesystemIterator::SKIP_DOTS));
foreach ($it as $f) {
    if (! str_ends_with($f->getPathname(), '.blade.php')) { continue; }
    $n++;
    try { token_get_all($compiler->compileString((string) file_get_contents($f->getPathname())), TOKEN_PARSE); }
    catch (Throwable $e) { $bad[] = str_replace(getcwd().'/', '', $f->getPathname()); }
}
$out['views_parsed'] = $n; $out['views_failed'] = $bad;

$out['classes'] = [
    'RotatingMailboxTransport' => class_exists(App\Mail\Transport\RotatingMailboxTransport::class),
    'MailboxRotation' => class_exists(App\Support\Mail\MailboxRotation::class),
    'AccountDeletionRequests' => class_exists(App\Support\Privacy\AccountDeletionRequests::class),
    'AccountDeletionRequestController' => class_exists(App\Http\Controllers\Auth\AccountDeletionRequestController::class),
    'RetentionDecisions' => class_exists(App\Support\Retention\RetentionDecisions::class),
    'revokeAllFor' => method_exists(App\Services\OAuthConnectionService::class, 'revokeAllFor'),
    'isComplimentary' => method_exists(App\Support\Billing\SubscriptionStatus::class, 'isComplimentary'),
];
$out['routes'] = [];
foreach (['admin.customers.extend-trial', 'admin.customers.complimentary', 'admin.customers.complimentary.end', 'app.security.deletion-request', 'owner.security.deletion-request'] as $r) {
    $out['routes'][$r] = Route::has($r);
}
// the buckets and the retention ruling, from CODE (no database)
$out['retention_ruling'] = App\Support\Privacy\RetentionCoverage::RULING;
$out['retention_decisions_registered'] = count(App\Support\Retention\RetentionDecisions::all());
$out['abandoned_bucket_defined'] = array_key_exists('abandoned', App\Support\Platform\PlatformCustomers::BUCKETS);
// mail: only the NAME of the active mailer and how many rotation mailboxes have a login (a count)
$out['mail_default'] = (string) config('mail.default');
$out['rotating_configured'] = is_array(config('mail.mailers.rotating'));
$out['rotation_mailboxes_with_login'] = count(App\Support\Mail\MailboxRotation::mailboxes());
$out['rotation_daily_limit'] = App\Support\Mail\MailboxRotation::dailyLimit();

$keys = [
    'admin/customers/show.extend_trial_heading', 'admin/customers/show.complimentary_heading', 'admin/customers/show.end_complimentary_button',
    'php/http/controllers/admin/customer_controller.trial_extended', 'php/support/platform/platform_customers.bucket_abandoned',
    'php/support/billing/subscription_status.complimentary_until', 'livewire/operator/billing.complimentary_line',
    'components/security/deletion-card.heading', 'components/security/deletion-dialog.send', 'admin/privacy/index.bucket_pruned',
    'admin/privacy/index.retention_intro', 'php/support/platform/health/health_board.mail_mailbox_usage', 'php/models/privacy_request.channel_in_app',
];
$missing = [];
$locales = array_values((array) config('app.available_locales'));
foreach ($locales as $l) { foreach ($keys as $k) { if (! Lang::has($k, $l, false)) { $missing[] = $l.':'.$k; } } }
$out['locales'] = count($locales); $out['texts_checked'] = count($keys); $out['texts_missing'] = $missing;
echo json_encode($out, JSON_UNESCAPED_SLASHES), PHP_EOL;
