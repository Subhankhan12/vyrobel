  _scrollTo(id) { const el = document.getElementById(id); if (!el) return; const y = window.scrollY + el.getBoundingClientRect().top - 20; window.scrollTo({ top: Math.max(0, y), behavior: 'smooth' }); }

  componentDidMount() { this._mount = Date.now(); this._cd = setInterval(() => this.setState(st => ({ tick: (st.tick || 0) + 1 })), 1000); }
  componentWillUnmount() { clearInterval(this._cd); }
  _pad(n) { return String(n).padStart(2, '0'); }

  renderVals() {
    const s = this.state;
    const step = s.step;
    const done = step === 4;
    const times = { '11': '11:00–13:00', '13': '13:00–15:00', '15': '15:00–18:00', '18': 'after 18:00' };

    const cdTotal = 24 * 86400000; // booking-to-arrival window
    const cdPct = Math.max(0, Math.min(100, (cdTotal - cdRem) / cdTotal * 100));
