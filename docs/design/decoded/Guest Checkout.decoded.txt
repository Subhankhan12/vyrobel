  SUBTOTAL = 501;

  eur(n, dec) { const d = (dec == null) ? 2 : dec; return '\u20ac' + Number(n).toLocaleString('en-IE', { minimumFractionDigits: d, maximumFractionDigits: d }); }

  walletBalance() {
    let refund = 48;
    try { const dd = JSON.parse(localStorage.getItem('vyrobel:refund:last')); if (dd && typeof dd.amount === 'number') refund = dd.amount; } catch (e) {}
    let promo = 0;
    try { const w = JSON.parse(localStorage.getItem('vyrobel_guest_wallet')); if (w && Array.isArray(w.added)) promo = w.added.reduce((a, t) => a + (Number(t.amt) || 0), 0); } catch (e) {}
    return refund + 25 + 20 - 15 + 10 + promo;
  }
  renderVals() {
    const bal = this.walletBalance();
    const hasCredit = bal > 0;
    const applied = this.state.applyCredit && hasCredit;
    const appliedAmt = applied ? Math.min(bal, this.SUBTOTAL) : 0;
    const total = this.SUBTOTAL - appliedAmt;
    return {
      hasCredit: hasCredit,
      applied: applied,
      creditSub: applied ? (this.eur(appliedAmt) + ' applied to this booking') : (this.eur(bal) + ' available in your wallet'),
